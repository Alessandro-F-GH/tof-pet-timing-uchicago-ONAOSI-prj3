\begin{block}{Detector system}
\begin{itemize}
  \item Two opposing LYSO--SiPM detector modules.
  \item LYSO crystal: $2\times2\times3~\mathrm{mm^3}$.
  \item BC SiPM: $4\times4~\mathrm{mm^2}$.
  \item FBK-HF front-end / amplifier.
  \item $^{22}$Na positron-emitting source centered between the two detectors.
\end{itemize}
\end{block}

\begin{block}{Analog outputs}
For both detectors:
\begin{itemize}
  \item \textbf{energy} channel: first stage of amplification.
  \item \textbf{timing} channel: second stage of amplification, designed for timing.
\end{itemize}
\end{block}
